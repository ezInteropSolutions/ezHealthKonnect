import asyncio
import aiohttp
import json
import os

# HL7 versions to process
#VERSIONS = ["2.1", "2.2", "2.3", "2.3.1", "2.4", "2.5", "2.5.1", "2.6", "2.7", "2.7.1", "2.8"]
VERSIONS = ["2.8"]
BASE_DIR = "C:/Projects/hl7schema/schemas"

# Cache for tables to avoid redundant API calls per session
table_cache = {}

async def fetch_json(session, url):
    """Fetch JSON data from API."""
    async with session.get(url) as response:
        response.raise_for_status()
        return await response.json()

async def fetch_table_values(session, base_url, table_id):
    """Fetch and cache HL7 table values."""
    if not table_id:
        return None
    if table_id in table_cache:
        return table_cache[table_id]

    url = f"{base_url}/Tables/{table_id}"
    print(f"         ↳ Fetching Table {table_id}")
    try:
        data = await fetch_json(session, url)
        entries = {entry["value"]: entry["description"] for entry in data.get("entries", [])}
        table_cache[table_id] = entries
        return entries
    except aiohttp.ClientResponseError:
        print(f"⚠️ Table {table_id} not found")
        return None

async def fetch_field_components(session, base_url, field_id):
    """Fetch subfields for complex fields."""
    url = f"{base_url}/Fields/{field_id}"
    print(f"      ↳ Fetching components for field {field_id}")
    try:
        data = await fetch_json(session, url)
        return data.get("fields", [])
    except aiohttp.ClientResponseError:
        return []

async def process_field(session, base_url, field):
    """Process a single field including subfields and table values."""
    field_schema = {
        "name": field["name"],
        "dataType": field["dataType"],
        "dataTypeName": field["dataTypeName"],
        "length": field["length"],
        "usage": field["usage"],
        "repeat": field["rpt"],
        "description": field.get("description", ""),
        "tableId": field.get("tableId"),
        "tableName": field.get("tableName"),
        "position": field.get("sequence"),  # 🆕 Capture position for field
    }
    if field.get("tableId"):
        field_schema["values"] = await fetch_table_values(session, base_url, field["tableId"])

    # Fetch subfields if complex
    if field["dataTypeName"] not in ["String Data", "Numeric", "Date", "Time"]:
        components = await fetch_field_components(session, base_url, field["id"])
        if components:
            subfields = {}
            for sub in components:
                sub_schema = {
                    "name": sub["name"],
                    "dataType": sub["dataType"],
                    "dataTypeName": sub["dataTypeName"],
                    "length": sub["length"],
                    "usage": sub["usage"],
                    "repeat": sub["rpt"],
                    "description": sub.get("description", ""),
                    "tableId": sub.get("tableId"),
                    "tableName": sub.get("tableName"),
                    "position": sub.get("sequence"),  # 🆕 Capture position for subcomponent
                }
                if sub.get("tableId"):
                    sub_schema["values"] = await fetch_table_values(session, base_url, sub["tableId"])
                subfields[sub["id"]] = sub_schema
            field_schema["components"] = subfields
    return field["id"], field_schema

async def process_segment(session, base_url, seg):
    """Process a single segment or group."""
    if seg["isGroup"]:
        print(f"   ↳ Group: {seg['name']} (processing child segments)")
        # Process child segments recursively
        child_tasks = [process_segment(session, base_url, child) for child in seg.get("segments", [])]
        child_segments = dict(await asyncio.gather(*child_tasks))
        return seg["name"], {
            "description": "Group of segments",
            "usage": seg["usage"],
            "repeat": seg["rpt"],
            "sequence": seg.get("sequence"),  # 🆕 Capture sequence for group
            "type": "group",
            "segments": child_segments
        }
    else:
        # Normal segment processing
        url = f"{base_url}/Segments/{seg['name']}"
        print(f"   ↳ Fetching fields for segment {seg['name']}")
        try:
            data = await fetch_json(session, url)
        except aiohttp.ClientResponseError as e:
            print(f"❌ Segment {seg['name']} not found: {e}")
            return seg["name"], {
                "description": "Segment not found in API",
                "usage": seg["usage"],
                "repeat": seg["rpt"],
                "sequence": seg.get("sequence"),  # 🆕 Capture sequence for segment
                "type": "segment",
                "fields": {}
            }

        # Process fields
        field_tasks = [process_field(session, base_url, field) for field in data.get("fields", [])]
        fields = dict(await asyncio.gather(*field_tasks))
        return seg["name"], {
            "description": data.get("description", ""),
            "usage": seg["usage"],
            "repeat": seg["rpt"],
            "sequence": seg.get("sequence"),  # 🆕 Capture sequence for segment
            "type": "segment",
            "fields": fields
        }

async def process_message(session, version, base_url, message_type, output_dir):
    """Build schema for a single message type."""
    url = f"{base_url}/TriggerEvents/{message_type}"
    print(f"📥 Fetching segments for {message_type}")
    data = await fetch_json(session, url)
    message_desc = data.get("description", "")
    segments = data.get("segments", [])

    segment_tasks = [process_segment(session, base_url, seg) for seg in segments]
    structure = dict(await asyncio.gather(*segment_tasks))

    schema = {
        "messageType": message_type,
        "description": message_desc,
        "version": version,
        "structure": structure
    }

    # Save to JSON
    os.makedirs(output_dir, exist_ok=True)
    output_file = os.path.join(output_dir, f"{message_type}.json")
    with open(output_file, "w", encoding="utf-8") as f:
        json.dump(schema, f, indent=2)
    print(f"✅ Saved {message_type} schema to {output_file}")

async def process_version(session, version):
    """Process all message types for a specific HL7 version."""
    base_url = f"https://hl7-definition.caristix.com/v2-api/1/HL7v{version}"
    output_dir = os.path.join(BASE_DIR, f"v{version}")
    print(f"🌐 Fetching all trigger events for version {version}")

    try:
        events_url = f"{base_url}/TriggerEvents"
        events_data = await fetch_json(session, events_url)
    except aiohttp.ClientResponseError as e:
        print(f"❌ Failed to fetch trigger events for v{version}: {e}")
        return

    message_types = [msg["id"] for msg in events_data if msg.get("type") == "Message"]
    print(f"Found {len(message_types)} message types for v{version}")

    # Process all message types concurrently
    message_tasks = [process_message(session, version, base_url, msg_type, output_dir) for msg_type in message_types]
    await asyncio.gather(*message_tasks)

async def main():
    async with aiohttp.ClientSession() as session:
        # Process each version sequentially
        for version in VERSIONS:
            table_cache.clear()  # Clear cache for each version
            await process_version(session, version)

if __name__ == "__main__":
    asyncio.run(main())
